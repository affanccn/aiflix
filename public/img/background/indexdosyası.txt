// Film önerisi fonksiyonu
function getMovieSuggestions() {
    let movieInput = document.getElementById("movieInput").value.trim();
    let numberOfSuggestions = document.getElementById("numberOfSuggestions").value;
    let genreSelect = document.getElementById("genreSelect").value;

    if (movieInput === "") {
        alert("Lütfen bir film adı girin.");
        return;
    }

    if (numberOfSuggestions < 1 || numberOfSuggestions > 10) {
        alert("Lütfen 1 ile 10 arasında bir sayı girin.");
        return;
    }

    // Seçilen kategoriye göre genre ID'sini al
    const genreIds = {
        action: 28,
        adventure: 12,
        horror: 27
    };

    const genreId = genreIds[genreSelect];

    function compareWithUserMovie(movieDescription) {
        let userMovieDescription = document.getElementById('movieInput').value.trim();
        
        fetch("https://api.openai.com/v1/completions", {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
                "Authorization": `Bearer ${openAI_API_KEY}`
            },
            body: JSON.stringify({
                model: "text-davinci-003",
                prompt: `The following is a comparison between two movies: one with the description "${userMovieDescription}" and the other with the description "${movieDescription}". Describe the similarities between the two movies.`,
                max_tokens: 150
            })
        })
        .then(response => response.json())
        .then(data => {
            let similarityText = data.choices[0].text.trim();
            displaySimilarities(similarityText);
        })
        .catch(error => console.error('Error comparing movies:', error));
    }
    
    function displaySimilarities(similarityText) {
        let suggestionsContainer = document.getElementById("suggestionsContainer");
        let similarityElement = document.createElement("div");
        similarityElement.classList.add("similarity-item");
        similarityElement.textContent = similarityText;
        suggestionsContainer.appendChild(similarityElement);
    }
    
    fetch(`${BASE_URL}/discover/movie?api_key=${API_KEY}&with_genres=${genreId}&sort_by=popularity.desc&page=1`)
        .then(response => response.json())
        .then(data => {
            let suggestions = [];
            for (let i = 0; i < numberOfSuggestions; i++) {
                let randomMovie = data.results[Math.floor(Math.random() * data.results.length)];
                suggestions.push(randomMovie);
            }
            displaySuggestions(suggestions);
        })
        .catch(error => console.error('Error fetching data:', error));
}

function getMovieDescription(movieName) {
    fetch(`${BASE_URL}/search/movie?api_key=${API_KEY}&query=${movieName}`)
        .then(response => response.json())
        .then(data => {
            if (data.results.length > 0) {
                let movieId = data.results[0].id;
                fetch(`${BASE_URL}/movie/${movieId}?api_key=${API_KEY}`)
                    .then(response => response.json())
                    .then(movieData => {
                        let movieDescription = movieData.overview;
                        console.log(movieDescription);
                        compareWithUserMovie(movieDescription);
                    });
            }
        })
        .catch(error => console.error('Error fetching movie details:', error));
}

// Film önerilerini ve nedenlerini ekleyen fonksiyon
function displaySuggestions(suggestions) {
    let suggestionsContainer = document.getElementById("suggestionsContainer");
    suggestionsContainer.innerHTML = "<h3>Önerilen Filmler:</h3>";

    suggestions.forEach(function(movie) {
        let movieElement = document.createElement("div");
        movieElement.classList.add("movie-item");

        // Film bilgilerini HTML formatında ekleyelim
        movieElement.innerHTML = `
            <h3>${movie.title}</h3>
            <img src="https://image.tmdb.org/t/p/w500${movie.poster_path}" alt="${movie.title}">
            <p><strong>Açıklama:</strong> ${movie.overview || "Açıklama bulunamadı."}</p>
            <p><strong>Benzerlik:</strong> ${getSimilarityExplanation(movie)}</p> <!-- Neden önerildiği burada -->
        `;

        suggestionsContainer.appendChild(movieElement);
    });
}

// Filmler arasındaki benzerlikleri açıklayan fonksiyon
function getSimilarityExplanation(movie) {
    let explanation = "Bu film, istediğiniz türde aksiyon, macera ve mücadele içeriyor. ";

    // Film türlerine göre açıklama ekleyebiliriz
    if (movie.genre_ids.includes(28)) { // Aksiyon türü ID'si
        explanation += "Aksiyon dolu sahneler içeriyor. ";
    }
    if (movie.genre_ids.includes(12)) { // Macera türü ID'si
        explanation += "Macera dolu bir hikayeye sahip. ";
    }
    if (movie.genre_ids.includes(27)) { // Korku türü ID'si
        explanation += "Korku dolu sahneler içeriyor. ";
    }

    // Film açıklamasına göre benzerlik ekleme
    if (movie.overview.includes("mücadele") || movie.overview.includes("savaş")) {
        explanation += "Kahramanın büyük bir mücadeleye girdiği bir hikaye var. ";
    }

    return explanation;
}
